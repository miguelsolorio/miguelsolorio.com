{{/* Contribution counts for the close of a project page. The inner content is one "value | label | icon" row per line, so the numbers stay next to the prose that frames them instead of moving into front matter. `icon` is optional and names one of the glyphs below; `source` is the "owner/repo" GitHub slug the counts came from, rendered as a link to that repository. */}} {{- $glyphs := dict "pr" `` "merge" `` "commit" `` "issue" `` "issue-closed" `` -}} {{- $rows := slice -}} {{- range (split .Inner "\n") -}} {{- $line := strings.Trim . " \t\r" -}} {{- with $line -}} {{- $parts := split . "|" -}} {{- if lt (len $parts) 2 -}} {{- errorf "stats shortcode in %s: %q is missing the \"value | label\" separator" $.Page.File.Path $line -}} {{- end -}} {{- $icon := "" -}} {{- if ge (len $parts) 3 -}} {{- $icon = strings.Trim (index $parts 2) " \t" -}} {{- if not (isset $glyphs $icon) -}} {{- errorf "stats shortcode in %s: unknown icon %q" $.Page.File.Path $icon -}} {{- end -}} {{- end -}} {{- $rows = $rows | append (dict "value" (strings.Trim (index $parts 0) " \t") "label" (strings.Trim (index $parts 1) " \t") "icon" $icon) -}} {{- end -}} {{- end -}}
{{- with .Get "source" }} {{- $slug := strings.Trim . " \t" }} {{- if not (findRE `^[\w.-]+/[\w.-]+$` $slug) }} {{- errorf "stats shortcode in %s: source %q must be an \"owner/repo\" GitHub slug" $.Page.File.Path $slug }} {{- end }}

{{ $slug }}

{{- end }}
{{- range $rows }}
{{- with .icon }} {{- end }}

{{- end }}
