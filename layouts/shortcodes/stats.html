{{/* Figures for the close of a project page. The inner content is one "value | label | icon" row per line, so the numbers stay next to the prose that frames them instead of moving into front matter. `icon` is optional and names one of the glyphs below: the first five carry GitHub contribution states, the rest component-library parity. `source` is also optional and only applies when the figures came from a repository — it takes an "owner/repo" slug and renders a link to it. */}} {{- $glyphs := dict "pr" `` "merge" `` "commit" `` "issue" `` "issue-closed" `` "library" `` "updated" `` "outdated" `` "missing" `` -}} {{- $arrows := dict "up" `` "down" `` -}} {{- $rows := slice -}} {{- range (split .Inner "\n") -}} {{- $line := strings.Trim . " \t\r" -}} {{- with $line -}} {{- $parts := split . "|" -}} {{- if lt (len $parts) 2 -}} {{- errorf "stats shortcode in %s: %q is missing the \"value | label\" separator" $.Page.File.Path $line -}} {{- end -}} {{- $icon := "" -}} {{- if ge (len $parts) 3 -}} {{- $icon = strings.Trim (index $parts 2) " \t" -}} {{- if not (isset $glyphs $icon) -}} {{- errorf "stats shortcode in %s: unknown icon %q" $.Page.File.Path $icon -}} {{- end -}} {{- end -}} {{- /* A leading sign turns the figure into a ticker row: the arrow renders the direction and the value keeps only its digits. Authors may write either a hyphen or a real minus, and the sign stays in the accessible name so the arrow is not the only thing carrying it. */ -}} {{- $value := strings.Trim (index $parts 0) " \t" -}} {{- $trend := "" -}} {{- if hasPrefix $value "+" -}} {{- $trend = "up" -}} {{- $value = strings.TrimPrefix "+" $value -}} {{- else if hasPrefix $value "\u2212" -}} {{- $trend = "down" -}} {{- $value = strings.TrimPrefix "\u2212" $value -}} {{- else if hasPrefix $value "-" -}} {{- $trend = "down" -}} {{- $value = strings.TrimPrefix "-" $value -}} {{- end -}} {{- $rows = $rows | append (dict "value" (strings.Trim $value " \t") "trend" $trend "label" (strings.Trim (index $parts 1) " \t") "icon" $icon) -}} {{- end -}} {{- end -}}
{{- with .Get "source" }} {{- $slug := strings.Trim . " \t" }} {{- if not (findRE `^[\w.-]+/[\w.-]+$` $slug) }} {{- errorf "stats shortcode in %s: source %q must be an \"owner/repo\" GitHub slug" $.Page.File.Path $slug }} {{- end }}

{{ $slug }}

{{- end }}
{{- range $rows }}
{{- with .icon }} {{- end }}
{{- with .trend }} {{ if eq . "up" }}up{{ else }}down{{ end }} {{- end }} {{ .value }} {{ .label }}
{{- end }}
