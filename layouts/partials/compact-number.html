{{- $value := float . -}} {{- $unit := "" -}} {{- $scaled := $value -}} {{- if ge $value 1000000.0 -}} {{- $unit = "M" -}} {{- $scaled = div $value 1000000.0 -}} {{- else if ge $value 1000.0 -}} {{- $unit = "k" -}} {{- $scaled = div $value 1000.0 -}} {{- end -}} {{- $display := printf "%.0f" $scaled -}} {{- if and (ne $unit "") (lt $scaled 100.0) -}} {{- $display = strings.TrimSuffix ".0" (printf "%.1f" $scaled) -}} {{- end -}} {{- printf "%s%s" $display $unit -}}
